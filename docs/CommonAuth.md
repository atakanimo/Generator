## Kural

1. **Auth çekirdeği her yerde Generator'ın.**
   - Ön yüzde [K20](klasorRefactor.md#4-generator-çıktısı): `<ad>.generated.ts` soneki, `Auto Generated By Generator - <tarih>` başlığı, elle düzenlenmez.
   - `services/api/` altındaki ortak dosyalar asset sync'in "ortak" sınıfıdır (http-layer.md).
   - Backend'de auth asset'leri asset sync'te "ortak" (aşağıda).
2. **Custom'ın yeri K20'nin soneksiz eşidir.**
   - Ön yüzde `<ad>.generated.ts`'in yanında `<ad>.ts` durur ve onu içeri alır. Normal modüllerdeki gibi: `create<E>Thunks(hooks)` → `createAuthThunks(hooks)`.
   - Backend'de `auth.hooks` ve `mailer`.
   - Generator K dosyayı yalnız yoksa oluşturur.
3. **UI üretilmez.**
   - Admin / web: `pages/Login/index.tsx` ([K19](klasorRefactor.md#3-ne-nereye-gider)). Admin'de `AtaLoginPage`.
   - Mobil: `screens/Auth/<Ad>/index.tsx`.
   - Guard'lar `routes/guards/` altında (`ProtectedRoute`, `RequireAuth`, [K14](klasorRefactor.md#3-ne-nereye-gider)).
   - Ekran yalnız thunk + selector kullanır.
4. **G dosyaya elle dokunulmaz.** İhtiyaç hook'a sığmıyorsa Generator'a hook eklenir (4 projeye birden); G dosya projede değiştirilmez.
5. **Users modeli** manifest'ten G (bugünkü gibi):
   - auth alanları kilitli;
   - projeye özel alanlar model editöründen;
   - özel uçlar custom bloklarda.

## Yazma yolu

Yeni bir komut yazılmaz; iki mevcut yol kullanılır.

```bash
node scripts/assets-sync.mjs <host ya da uygulama klasörü>                  # özet: eksik / farklı / aynı / proje
node scripts/assets-sync.mjs <klasör> --diff [yol…]
node scripts/assets-sync.mjs <klasör> --apply yol [yol…] | --apply-all
```

- **Backend auth:** asset sync'e auth asset'leri eklenir, yukarıdaki ortak / proje tablosuyla. Modül koşulları manifest'ten okunur.
- **`services/api/auth.ts` + `platform.ts`:** asset sync (client-http). `auth.ts` "proje"den "ortak"a geçer. Projede farklıysa `--diff` ile gösterilir, sessizce ezilmez.
- **İstemci G / K dosyaları:** client data layer generate'i (`generateClientModules`, kapı `regen-client.mjs`). Admin de buna girer (`CLIENT_REDUX_PRESET` admin → `redux-web` + `admin: true`).
- Sonra Electron Generate ekranında Auth hedefi (client-data-layer C4 ile birlikte).

### A1. `src/services/auth.service.generated.ts` (G)

```ts
// Auto Generated By Generator - <tarih>
// Ortak auth uçları (docs/backend/auth.md "Sözleşme"). Elle düzenlenmez: değişiklik Generator'da yapılır.
import { API } from './api';
import type { LocalResponse } from '@/types';

export interface AuthUser {
  id: string;
  email: string;
  name: string;
  role: 'admin' | 'user';
  provider: 'password' | 'google' | 'apple';
  emailVerified: boolean;
  hasPassword: boolean;
}

/** `login` / `refresh` cevabı. Host'un `onSession` alanları (ör. Varlık `profile`) aynı nesnede gelir. */
export type AuthSession = {
  accessToken: string;
  refreshToken: string;
  /** Access token ömrü (saniye). */
  expiresIn: number;
  user: AuthUser;
  isNewUser?: boolean;
} & Record<string, unknown>;

export interface LoginBody { email: string; password: string }
export interface ChangePasswordBody { currentPassword: string; newPassword: string }
export interface AdminSetPasswordBody { userId: string; newPassword: string }

/** Önek manifest'ten: Express / .NET `/api`, Python `/api/v1`. */
export const AUTH_ENDPOINTS = {
  login: '/api/auth/login',
  refresh: '/api/auth/refresh',
  logout: '/api/auth/logout',
  me: '/api/auth/me',
  changePassword: '/api/auth/change-password',
  adminSetPassword: '/api/auth/admin/set-password'
} as const;

export const login = (data: LoginBody): Promise<LocalResponse<AuthSession>> =>
  API({ url: AUTH_ENDPOINTS.login, method: 'POST', data, skipSuccessToast: true });

export const me = (): Promise<LocalResponse<AuthUser>> =>
  API({ url: AUTH_ENDPOINTS.me, method: 'GET', skipLoading: true });

/** O oturumun refresh token'ını iptal eder. */
export const logout = (refreshToken: string): Promise<LocalResponse<null>> =>
  API({ url: AUTH_ENDPOINTS.logout, method: 'POST', data: { refreshToken }, skipLoading: true, skipSuccessToast: true });

export const changePassword = (data: ChangePasswordBody): Promise<LocalResponse<null>> =>
  API({ url: AUTH_ENDPOINTS.changePassword, method: 'POST', data });

export const adminSetPassword = (data: AdminSetPasswordBody): Promise<LocalResponse<null>> =>
  API({ url: AUTH_ENDPOINTS.adminSetPassword, method: 'POST', data });
```

Refresh burada yok. Interceptor'sız çağrılması gerektiği için ortak `services/api/auth.ts`'te (A4).

// Auth thunk fabrikası. Thunk'lar hep fulfilled döner; sayfa `unwrap()` sonrası `isSuccess` ile dallanır.
// Projeye özel davranış hook'larla `store/thunks/auth.thunk.ts`'ten verilir. Elle düzenlenmez.
import { createAsyncThunk, type Dispatch } from '@reduxjs/toolkit';
import type { LocalResponse } from '@/types';
import * as service from '@/services/auth.service.generated';
import type { AuthSession, AuthUser, ChangePasswordBody, LoginBody } from '@/services/auth.service.generated';
import { authStorage } from '@/services/api/auth';

/** `deniedKey`: `canSignIn`'in reddi (i18n anahtarı); sayfa onu gösterir. Sunucu hatasını interceptor toast'lar. */
export type AuthResult<T> = LocalResponse<T> & { deniedKey?: string };

export interface AuthThunkHooks {
  /** `null` → girebilir; string → reddedilir, oturum kapanır, refresh BE'de iptal edilir. */
  canSignIn?: (user: AuthUser) => string | null;
  /** login, refresh ve açılıştaki `me` sonrası: ek alanları başka slice'lara yazmak için. */
  onSession?: (session: AuthSession, dispatch: Dispatch) => void;
  /** logout ve oturum bitince: diğer slice'ları sıfırlamak için. */
  onSignedOut?: (dispatch: Dispatch) => void;
}

const failed = <T>(error: unknown): AuthResult<T> =>
  error && typeof error === 'object' && 'isSuccess' in error
    ? (error as AuthResult<T>)
    : ({ isSuccess: false, data: null, uiMessage: '', errorData: null } as unknown as AuthResult<T>);

const denied = <T>(deniedKey: string): AuthResult<T> =>
  ({ isSuccess: false, data: null, uiMessage: '', errorData: null, deniedKey }) as unknown as AuthResult<T>;

export const createAuthThunks = ({ canSignIn, onSession, onSignedOut }: AuthThunkHooks = {}) => {
  /** `remember`: token'lar `localStorage`'a, yoksa `sessionStorage`'a. BE'ye gitmez. */
  const login = createAsyncThunk('auth/login', async ({ remember, ...body }: LoginBody & { remember?: boolean }, { dispatch }): Promise<AuthResult<AuthSession>> => {
    try {
      const response = await service.login(body);
      if (!response.isSuccess || !response.data) return response;
      const deniedKey = canSignIn?.(response.data.user) ?? null;
      if (deniedKey) {
        service.logout(response.data.refreshToken).catch(() => undefined);
        return denied(deniedKey);
      }
      await authStorage.save(response.data.accessToken, response.data.refreshToken, remember);
      onSession?.(response.data, dispatch);
      return response;
    } catch (error) {
      return failed(error);
    }
  });

  /** Açılış: kayıtlı oturum `me` ile doğrulanır; süresi dolmuşsa interceptor yeniler. */
  const restoreSession = createAsyncThunk('auth/restoreSession', async (_: void, { dispatch }): Promise<AuthResult<AuthUser> & { accessToken: string | null }> => {
    const stored = await authStorage.read();
    if (!stored.accessToken && !stored.refreshToken) return { ...denied<AuthUser>(''), accessToken: null };
    try {
      const response = await service.me();
      const deniedKey = response.data ? canSignIn?.(response.data) ?? null : null;
      if (deniedKey) {
        await authStorage.clear();
        onSignedOut?.(dispatch);
        return { ...denied<AuthUser>(deniedKey), accessToken: null };
      }
      return { ...response, accessToken: (await authStorage.read()).accessToken };
    } catch (error) {
      return { ...failed<AuthUser>(error), accessToken: null };
    }
  });

  /** Sunucu cevabı ne olursa olsun yerel oturum kapanır. */
  const logout = createAsyncThunk('auth/logout', async (_: void, { dispatch }): Promise<AuthResult<null>> => {
    const { refreshToken } = await authStorage.read();
    try {
      if (refreshToken) await service.logout(refreshToken);
    } catch {
      // Ağ hatası: yerel çıkış yine yapılır.
    }
    await authStorage.clear();
    onSignedOut?.(dispatch);
    return { isSuccess: true, data: null, uiMessage: '', errorData: null } as AuthResult<null>;
  });

  /** BE kullanıcının bütün oturumlarını kapatır: yeni şifreyle yeniden girilir. */
  const changePassword = createAsyncThunk('auth/changePassword', async (body: ChangePasswordBody, { dispatch }): Promise<AuthResult<null>> => {
    try {
      const response = await service.changePassword(body);
      if (response.isSuccess) {
        await authStorage.clear();
        onSignedOut?.(dispatch);
      }
      return response;
    } catch (error) {
      return failed(error);
    }
  });

  return { login, restoreSession, logout, changePassword };
};

export type AuthThunks = ReturnType<typeof createAuthThunks>;
```

// bitmediğini buradan öğrenir. Token saklama ve tek seferlik refresh de burada. Generator'ın; projede değiştirilmez.
import axios from 'axios';
import type { ApiAuth } from '@/types';
import { APP_CONFIG } from '@/config/app';
import { AUTH_ENDPOINTS, type AuthSession } from '../auth.service.generated';
import { BASE_URL, tokenStore } from './platform';

const ACCESS_KEY = `${APP_CONFIG.storagePrefix}.access_token`;
const REFRESH_KEY = `${APP_CONFIG.storagePrefix}.refresh_token`;

export const authStorage = {
  read: async () => ({ accessToken: await tokenStore.get(ACCESS_KEY), refreshToken: await tokenStore.get(REFRESH_KEY) }),
  /** `remember` verilmezse token'lar şu an bulundukları yerde kalır (refresh). */
  save: async (accessToken: string, refreshToken: string, remember?: boolean) => {
    const persistent = remember ?? (await tokenStore.isPersistent(REFRESH_KEY));
    await tokenStore.set(ACCESS_KEY, accessToken, persistent);
    await tokenStore.set(REFRESH_KEY, refreshToken, persistent);
  },
  clear: async () => {
    await tokenStore.remove(ACCESS_KEY);
    await tokenStore.remove(REFRESH_KEY);
  }
};

type AuthState = { auth?: { accessToken?: string | null } };
const accessTokenOf = (state: unknown) => (state as AuthState | undefined)?.auth?.accessToken ?? null;

let inFlight: Promise<boolean> | null = null;

/** Aynı anda gelen 401'ler tek isteği bekler. Ağ hatasında oturum kalır; sunucu reddederse kapanır. */
const refreshOnce = async (dispatch: (action: unknown) => unknown): Promise<boolean> => {
  const { sessionChanged, tokenAdopted } = await import('@/store/slices/auth.slice');
  const sent = (await authStorage.read()).refreshToken;
  if (!sent) {
    dispatch(sessionChanged(null));
    return false;
  }
  try {
    const res = await axios.post(AUTH_ENDPOINTS.refresh, { refreshToken: sent }, { baseURL: BASE_URL });
    const session = res.data.data as AuthSession;
    await authStorage.save(session.accessToken, session.refreshToken);
    dispatch(sessionChanged(session));
    return true;
  } catch (error) {
    // Başka sekme aynı anda yeniledi: onun token'ları storage'da.
    const stored = await authStorage.read();
    if (stored.refreshToken && stored.accessToken && stored.refreshToken !== sent) {
      dispatch(tokenAdopted(stored.accessToken));
      return true;
    }
    if (!(axios.isAxiosError(error) && !error.response)) {
      await authStorage.clear();
      dispatch(sessionChanged(null));
    }
    return false;
  }
};

const apiAuth: ApiAuth = {
  getAccessToken: async state => accessTokenOf(state) || (await authStorage.read()).accessToken,
  hasSession: state => Boolean(accessTokenOf(state)),
  refreshSessionOnce: dispatch => (inFlight ??= refreshOnce(dispatch).finally(() => (inFlight = null))),
  isAuthRequest: url => {
    const path = (url ?? '').split('?')[0];
    return [AUTH_ENDPOINTS.login, AUTH_ENDPOINTS.refresh, AUTH_ENDPOINTS.logout].some(endpoint => path.endsWith(endpoint));
  }
};

export const { getAccessToken, hasSession, refreshSessionOnce, isAuthRequest } = apiAuth;

/** Token saklama (browser): `persistent` → localStorage ("Beni hatırla"), değilse sessionStorage. Mobilde AsyncStorage. */
const safe = <T>(fn: () => T, fallback: T): T => {
  try {
    return fn();
  } catch {
    return fallback; // gizli pencere / engelli storage: oturum yalnız bellekte
  }
};

export const tokenStore = {
  get: async (key: string) => safe(() => localStorage.getItem(key) ?? sessionStorage.getItem(key), null),
  set: async (key: string, value: string, persistent: boolean) =>
    safe(() => {
      (persistent ? localStorage : sessionStorage).setItem(key, value);
      (persistent ? sessionStorage : localStorage).removeItem(key);
    }, undefined),
  remove: async (key: string) => safe(() => (localStorage.removeItem(key), sessionStorage.removeItem(key)), undefined),
  isPersistent: async (key: string) => safe(() => localStorage.getItem(key) !== null, false)
};
```
